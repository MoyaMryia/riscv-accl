def merge_intervals(busy):
    """
    Merge overlapping or touching intervals from a list of half-open intervals.
    
    Parameters:
    - busy: list of tuples (a, b) representing half-open intervals [a, b)
    
    Returns:
    - list of sorted tuples representing merged intervals
    
    Raises:
    - ValueError: if any interval is reversed (start > end)
    """
    if not busy:
        return []
    
    # Validate all intervals and sort by start time
    valid = []
    for a, b in busy:
        if a > b:
            raise ValueError("Intervals must be non-reversed")
        valid.append((a, b))
    
    # Sort by start time
    valid.sort(key=lambda x: x[0])
    
    merged = []
    i = 0
    n = len(valid)
    
    while i < n:
        current = valid[i]
        merged.append(current)
        
        i += 1
        j = i + 1
        
        while j < n:
            next_interval = valid[j]
            
            # Check if intervals overlap or touch
            if next_interval[0] <= current[1]:
                # Merge intervals
                current = (current[0], max(current[1], next_interval[1]))
                j += 1
            else:
                break
        
        i = j
    
    return merged

def free_windows(busy, start, end):
    """
    Find maximal free intervals within a query window [start, end).
    
    Parameters:
    - busy: list of tuples (a, b) representing busy intervals
    - start: float, start of the query window
    - end: float, end of the query window
    
    Returns:
    - list of tuples representing maximal free intervals
    
    Raises:
    - ValueError: if end < start
    """
    if end < start:
        raise ValueError("Query window must have start <= end")
    
    # Filter busy intervals to those within the query window
    # We only care about intervals that overlap with [start, end)
    # An interval [a, b) overlaps if a < end and b > start
    clipped = []
    for a, b in busy:
        if a < end and b > start:
            clipped.append((a, b))
    
    # Sort by start time
    clipped.sort(key=lambda x: x[0])
    
    # Merge overlapping intervals
    merged = []
    i = 0
    n = len(clipped)
    
    while i < n:
        current = clipped[i]
        merged.append(current)
        
        i += 1
        j = i + 1
        
        while j < n:
            next_interval = clipped[j]
            
            # Check if intervals overlap
            if next_interval[0] <= current[1]:
                # Merge intervals
                current = (current[0], max(current[1], next_interval[1]))
                j += 1
            else:
                break
        
        i = j
    
    # Convert to half-open intervals for output
    result = []
    for a, b in merged:
        result.append((a, b))
    
    return result
