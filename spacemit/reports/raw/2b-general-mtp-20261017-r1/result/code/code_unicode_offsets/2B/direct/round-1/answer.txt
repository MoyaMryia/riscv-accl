def run_spans(text):
    """
    Return a list of (character, start, end) tuples for maximal consecutive runs of Unicode code points.
    Uses zero-based half-open offsets.
    """
    if not isinstance(text, str):
        raise ValueError("Input must be a string")
    
    if not text:
        return []
    
    result = []
    current_char = None
    current_start = 0
    current_end = 0
    
    for i, c in enumerate(text):
        if c == current_char:
            current_end = i + 1
        else:
            if current_char is not None:
                result.append((current_char, current_start, current_end))
            current_char = c
            current_start = i
            current_end = i + 1
    
    if current_char is not None:
        result.append((current_char, current_start, current_end))
    
    return result


def expand_spans(spans):
    """
    Reconstructs the original text from a list of spans.
    Raises ValueError if any span is malformed.
    """
    if not spans:
        return ""
    
    # Validate each span
    for span in spans:
        if not isinstance(span, (list, tuple)):
            raise ValueError("Every span must be a three-element list or tuple")
        if len(span) != 3:
            raise ValueError("Each span must have exactly 3 elements")
        char, start, end = span
        
        # Check character type
        if not isinstance(char, str) or len(char) != 1:
            raise ValueError("Each span must be a one-character string")
        
        # Check integer types
        if not isinstance(start, int) or not isinstance(end, int):
            raise ValueError("Start and end must be integers")
        
        # Check boolean exclusion
        if isinstance(start, bool) or isinstance(end, bool):
            raise ValueError("Start and end must not be bool")
        
        # Check non-negative start
        if start < 0:
            raise ValueError("Start must be >= 0")
        
        # Check end > start
        if end <= start:
            raise ValueError("End must be > start")
        
        # Check no gaps or overlaps (consecutive)
        if not (start == 0 or start == spans[-1][1] + 1):
            raise ValueError("Spans must be consecutive with no gaps")
    
    # Reconstruct text
    result = []
    current_start = 0
    current_end = 0
    
    for span in spans:
        char, start, end = span
        
        # Validate character bounds
        if not isinstance(char, str) or len(char) != 1:
            raise ValueError("Each span must be a one-character string")
        
        if not isinstance(start, int) or not isinstance(end, int):
            raise ValueError("Start and end must be integers")
        
        if isinstance(start, bool) or isinstance(end, bool):
            raise ValueError("Start and end must not be bool")
        
        if start < 0:
            raise ValueError("Start must be >= 0")
        
        if end <= start:
            raise ValueError("End must be > start")
        
        # Validate no gaps or overlaps
        if not (start == 0 or start == current_end + 1):
            raise ValueError("Spans must be consecutive with no gaps")
        
        # Append character
        result.append(char)
        current_end = end
    
    return "".join(result)
